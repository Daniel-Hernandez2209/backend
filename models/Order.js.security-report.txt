## Análisis de Vulnerabilidades de Ciberseguridad

He identificado varias vulnerabilidades críticas en tu modelo de pedidos. Aquí está el análisis detallado:

## 🚨 **VULNERABILIDADES CRÍTICAS**

### 1. **Inyección NoSQL (Líneas 290-297)**
**Severidad: CRÍTICA**

```js
// VULNERABLE - Línea 290-297
orderSchema.statics.searchOrders = function(query) {
  const searchRegex = new RegExp(query, 'i'); // ❌ Input sin sanitizar
  
  return this.find({
    $or: [
      { orderNumber: searchRegex },
      // ...
    ]
  });
};
```

**Riesgo:** Un atacante puede inyectar expresiones regulares maliciosas para causar ReDoS (Regular Expression Denial of Service) o extraer datos.

**Corrección:**
```js
// ✅ SEGURO
orderSchema.statics.searchOrders = function(query) {
  // Sanitizar y validar entrada
  if (!query || typeof query !== 'string') {
    return this.find({});
  }
  
  // Escapar caracteres especiales de regex
  const sanitizedQuery = query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&');
  const searchRegex = new RegExp(sanitizedQuery, 'i');
  
  return this.find({
    $or: [
      { orderNumber: searchRegex },
      { 'shippingAddress.firstName': searchRegex },
      { 'shippingAddress.lastName': searchRegex },
      { 'guestInfo.firstName': searchRegex },
      { 'guestInfo.lastName': searchRegex },
      { 'guestInfo.email': searchRegex }
    ]
  }).populate('user', 'firstName lastName email')
    .populate('items.product', 'name images sku')
    .sort({ createdAt: -1 });
};
```

### 2. **Falta de Validaciones de Entrada Críticas**
**Severidad: ALTA**

**Problema:** Múltiples campos carecen de validación adecuada:

```js
// ❌ VULNERABLE - Sin validaciones
guestInfo: {
  email: String,          // Sin validación de formato
  firstName: String,      // Sin longitud máxima
  lastName: String,       // Sin caracteres permitidos
  phone: String          // Sin validación de formato
},
```

**Corrección:**
```js
// ✅ SEGURO
const validator = require('validator');

guestInfo: {
  email: {
    type: String,
    validate: {
      validator: function(email) {
        return !email || validator.isEmail(email);
      },
      message: 'Email inválido'
    },
    maxlength: [254, 'Email demasiado largo']
  },
  firstName: {
    type: String,
    maxlength: [50, 'Nombre demasiado largo'],
    validate: {
      validator: function(name) {
        return !name || /^[a-zA-ZáéíóúüñÁÉÍÓÚÜÑ\s]+$/.test(name);
      },
      message: 'Nombre contiene caracteres inválidos'
    }
  },
  lastName: {
    type: String,
    maxlength: [50, 'Apellido demasiado largo'],
    validate: {
      validator: function(name) {
        return !name || /^[a-zA-ZáéíóúüñÁÉÍÓÚÜÑ\s]+$/.test(name);
      },
      message: 'Apellido contiene caracteres inválidos'
    }
  },
  phone: {
    type: String,
    validate: {
      validator: function(phone) {
        return !phone || /^[\+]?[0-9\s\-\(\)]{7,20}$/.test(phone);
      },
      message: 'Teléfono inválido'
    }
  }
},
```

### 3. **Potencial XSS en Campos de Texto (Líneas 134-138, 145-149)**
**Severidad: MEDIA**

```js
// ❌ VULNERABLE
notes: {
  customer: String,    // Sin sanitización
  admin: String,      // Sin sanitización  
  delivery: String    // Sin sanitización
},
notifications: [{
  content: String,    // Sin sanitización
  // ...
}]
```

**Corrección:**
```js
const DOMPurify = require('isomorphic-dompurify');

// Middleware pre-save para sanitizar
orderSchema.pre('save', function(next) {
  // Sanitizar notas
  if (this.notes.customer) {
    this.notes.customer = DOMPurify.sanitize(this.notes.customer, { ALLOWED_TAGS: [] });
  }
  if (this.notes.admin) {
    this.notes.admin = DOMPurify.sanitize(this.notes.admin, { ALLOWED_TAGS: [] });
  }
  if (this.notes.delivery) {
    this.notes.delivery = DOMPurify.sanitize(this.notes.delivery, { ALLOWED_TAGS: [] });
  }
  
  // Sanitizar contenido de notificaciones
  this.notifications.forEach(notification => {
    if (notification.content) {
      notification.content = DOMPurify.sanitize(notification.content, { ALLOWED_TAGS: [] });
    }
  });
  
  next();
});
```

### 4. **Exposición de Datos Sensibles (Líneas 180-186)**
**Severidad: MEDIA**

```js
// ❌ VULNERABLE - Información de pago sensible
payment: {
  transactionId: String,     // Datos sensibles sin cifrar
  pseReference: String,      // Datos sensibles sin cifrar
  // ...
}
```

**Corrección:**
```js
const crypto = require('crypto');

// Cifrado de datos sensibles
orderSchema.pre('save', function(next) {
  if (this.payment.transactionId && this.isModified('payment.transactionId')) {
    this.payment.transactionId = this.encryptSensitiveData(this.payment.transactionId);
  }
  if (this.payment.pseReference && this.isModified('payment.pseReference')) {
    this.payment.pseReference = this.encryptSensitiveData(this.payment.pseReference);
  }
  next();
});

orderSchema.methods.encryptSensitiveData = function(data) {
  const algorithm = 'aes-256-gcm';
  const key = process.env.ENCRYPTION_KEY; // 32 bytes key
  const iv = crypto.randomBytes(16);
  
  const cipher = crypto.createCipher(algorithm, key);
  cipher.setAAD(Buffer.from('order-data', 'utf8'));
  
  let encrypted = cipher.update(data, 'utf8', 'hex');
  encrypted += cipher.final('hex');
  const authTag = cipher.getAuthTag();
  
  return `${iv.toString('hex')}:${authTag.toString('hex')}:${encrypted}`;
};
```
