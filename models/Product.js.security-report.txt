## Análisis de Vulnerabilidades de Ciberseguridad

He identificado varias vulnerabilidades importantes en tu modelo de productos de Node.js/Mongoose:

### 🚨 **VULNERABILIDADES CRÍTICAS**

#### 1. **Inyección NoSQL** (Líneas 127-138)
**Problema**: El método `searchProducts` es vulnerable a inyección NoSQL.
```js
// VULNERABLE - Línea 127-138
productSchema.statics.searchProducts = function(query, category = null, limit = 20) {
  const searchCriteria = {
    isActive: true,
    $text: { $search: query }  // ⚠️ Sin validación
  };
  
  if (category) {
    searchCriteria.category = category;  // ⚠️ Sin validación
  }
```

**Corrección:**
```js
const validator = require('validator');

productSchema.statics.searchProducts = function(query, category = null, limit = 20) {
  // Validar y sanitizar entradas
  if (!query || typeof query !== 'string') {
    throw new Error('Query debe ser un string válido');
  }
  
  // Sanitizar query para prevenir inyección
  const sanitizedQuery = validator.escape(query.trim());
  
  const searchCriteria = {
    isActive: true,
    $text: { $search: sanitizedQuery }
  };
  
  // Validar categoría contra enum permitido
  const allowedCategories = ['hombre', 'mujer', 'deportivos', 'hoodies-sacos', 'chaquetas'];
  if (category) {
    if (!allowedCategories.includes(category)) {
      throw new Error('Categoría no válida');
    }
    searchCriteria.category = category;
  }
  
  // Validar y limitar el parámetro limit
  const safeLimit = Math.min(Math.max(parseInt(limit) || 20, 1), 100);
  
  return this.find(searchCriteria)
    .limit(safeLimit)
    .sort({ score: { $meta: 'textScore' } });
};
```

```js

    type: String,
    required: true  // ⚠️ Sin validación de URL
```

#### 3. **Inyección de Comandos en Generación de Slug** (Líneas 114-121)
**Problema**: El slug se genera sin sanitización adecuada.

```js
// VULNERABLE - Líneas 114-121
productSchema.pre('save', function(next) {
  if (this.isModified('name') && !this.slug) {
    this.slug = this.name
      .toLowerCase()
      .replace(/[^a-zA-Z0-9\s]/g, '')  // ⚠️ Sanitización insuficiente
      .replace(/\s+/g, '-');
  }
  next();
});
```

**Corrección:**
```js
const slugify = require('slugify');

productSchema.pre('save', function(next) {
  if (this.isModified('name') && !this.slug) {
    // Usar librería especializada y validar entrada
    if (!this.name || typeof this.name !== 'string') {
      return next(new Error('Nombre de producto inválido'));
    }
    
    this.slug = slugify(this.name, {
      lower: true,
      strict: true, // Elimina caracteres especiales
      remove: /[*+~.()'"!:@]/g // Elimina caracteres peligrosos
    });
    
    // Validar que el slug generado sea válido
    if (!this.slug || this.slug.length < 1) {
      return next(new Error('No se pudo generar un slug válido'));
    }
  }
  next();
});
```
