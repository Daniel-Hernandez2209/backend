Como experto en ciberseguridad de Node.js, he identificado varias vulnerabilidades importantes en tu código. Aquí está mi análisis detallado:

## 🚨 VULNERABILIDADES CRÍTICAS

### 1. **EXPOSICIÓN DE DATOS SENSIBLES**

**Líneas 94-96 y 168-170:** Exposición de información del error y stack trace
```js
// VULNERABLE
details: process.env.NODE_ENV === 'development' ? error.message : undefined

// VULNERABLE  
stack: process.env.NODE_ENV === 'development' ? error.stack : undefined
```

**Riesgo:** Alto - Revela información interna del sistema
**Solución:**
```js
// SEGURO - Nunca exponer detalles internos
return res.status(503).json({
  success: false,
  message: "Servicio temporalmente no disponible"
  // Remover completamente 'details' y 'stack'
});
```

### 2. **INYECCIÓN NoSQL**

**Líneas 207-208:** El endpoint de test permite inserción directa sin validación
```js
// VULNERABLE
const doc = await TestModel.create({ name: "Hola Atlas desde Vercel" });
```

**Riesgo:** Alto - Permite inyección NoSQL si se parametriza
**Solución:**
```js
// SEGURO - Con validación y sanitización
const { body, validationResult } = require('express-validator');

app.get("/api/test-db", [
  // Validar y sanitizar entrada si fuera necesario
], async (req, res) => {
  try {
    // Usar datos estáticos o validados
    const doc = await TestModel.create({ 
      name: "Test Connection",
      timestamp: new Date()
    });
    
    res.json({
      success: true,
      message: "Conexión exitosa",
      timestamp: doc.createdAt
      // NO exponer documentos completos
    });
  } catch (error) {
    console.error("Error en test-db:", error);
    res.status(500).json({ 
      success: false, 
      message: "Error interno del servidor"
    });
  }
});
```

### 3. **FALTA DE VALIDACIONES DE ENTRADA**

**Líneas 68-74:** Sin validación de entrada en JSON y URL
```js
// VULNERABLE - Sin validación
app.use(express.json({ 
  limit: '10mb',
  verify: (req, res, buf) => {
    req.rawBody = buf;
  }
}));
```

**Solución:**
```js
// SEGURO - Con validación y límites estrictos
const { body, param, query, validationResult } = require('express-validator');

app.use(express.json({ 
  limit: '1mb', // Reducir límite
  verify: (req, res, buf) => {
    try {
      JSON.parse(buf);
      req.rawBody = buf;
    } catch (error) {
      const err = new Error('JSON inválido');
      err.status = 400;
      throw err;
    }
  }
}));

// Middleware de validación global
const validateRequest = (req, res, next) => {
  const errors = validationResult(req);
  if (!errors.isEmpty()) {
    return res.status(400).json({
      success: false,
      message: 'Datos inválidos',
      errors: errors.array()
    });
  }
  next();
};
```

### 4. **CONFIGURACIÓN DE CORS INSEGURA**

**Líneas 41-43:** Permite cualquier origen cuando origin es undefined
```js
// VULNERABLE
if (!origin) return callback(null, true);
```

**Solución:**
```js
// SEGURO - Más restrictivo
const corsOptions = {
  origin: function (origin, callback) {
    const allowedOrigins = [
      'https://athenabrand.co',
      'https://www.athenabrand.co'
    ];
    
    // Solo permitir orígenes específicos en producción
    if (process.env.NODE_ENV === 'development') {
      allowedOrigins.push('http://localhost:4200', 'http://localhost:3000');
    }
    
    if (allowedOrigins.indexOf(origin) !== -1) {
      callback(null, true);
    } else {
      callback(new Error('No permitido por CORS'));
    }
  },
  credentials: true,
  optionsSuccessStatus: 200
};
```

### 5. **EXPOSICIÓN DE INFORMACIÓN DEL SISTEMA**

**Líneas 99-108:** El health check expone demasiada información
```js
// VULNERABLE - Expone información interna
uptime: process.uptime(),
environment: process.env.NODE_ENV,
database: mongoose.connection.readyState === 1 ? 'Conectado' : 'Desconectado'
```

**Solución:**
```js
// SEGURO - Información mínima
app.get('/health', (req, res) => {
  res.status(200).json({
    status: 'OK',
    timestamp: new Date().toISOString()
  });
});
```

## 📋 RECOMENDACIONES ADICIONALES

### 6. **Implementar Content Security Policy (CSP)**
```js
app.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'"],
      scriptSrc: ["'self'"],
      imgSrc: ["'self'", "data:", "https:"],
    },
  },
  crossOriginResourcePolicy: { policy: "cross-origin" }
}));
```

### 7. **Sanitización de datos**
```js
const mongoSanitize = require('express-mongo-sanitize');
const xss = require('xss-clean');

app.use(mongoSanitize()); // Prevenir inyección NoSQL
app.use(xss()); // Prevenir XSS
```

### 8. **Rate limiting más estricto**
```js
const authLimiter = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 5, // Más restrictivo para autenticación
  message: 'Demasiados intentos de login'
});

app.use('/api/auth/login', authLimiter);
```
