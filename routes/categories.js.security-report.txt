## Análisis de Vulnerabilidades - routes/categories.js

### ✅ **Aspectos Positivos Detectados:**
- Implementación de middleware de autenticación (`adminAuth`) para rutas administrativas
- Separación clara entre rutas públicas y administrativas
- Uso de controladores externos (separación de responsabilidades)

### ⚠️ **Vulnerabilidades y Recomendaciones:**

#### 1. **Falta de Validación de Entrada** - Líneas 11, 13, 14, 16, 17, 19
**Riesgo:** Alto - Los parámetros `:slug` no están validados

```js
// PROBLEMÁTICO - Sin validación
router.get('/:slug', CategoryController.getCategoryBySlug);
```

**Solución:**
```js
const { body, param, validationResult } = require('express-validator');

// Middleware de validación
const validateSlug = [
  param('slug')
    .isSlug()
    .isLength({ min: 1, max: 100 })
    .matches(/^[a-z0-9-]+$/)
    .withMessage('Slug debe contener solo letras minúsculas, números y guiones'),
  (req, res, next) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }
    next();
  }
];

// Aplicar validación
router.get('/:slug', validateSlug, CategoryController.getCategoryBySlug);
router.get('/seo/:slug', validateSlug, CategoryController.getCategorySEO);
router.get('/:slug/subcategories', validateSlug, CategoryController.getSubcategories);
router.put('/:slug/toggle', adminAuth, validateSlug, CategoryController.toggleCategory);
```

#### 2. **Falta de Rate Limiting** - Todas las líneas
**Riesgo:** Medio - Posibles ataques de denegación de servicio

**Solución:**
```js
const rateLimit = require('express-rate-limit');

// Rate limiting para rutas públicas
const publicRateLimit = rateLimit({
  windowMs: 15 * 60 * 1000, // 15 minutos
  max: 100, // máximo 100 requests por IP
  message: 'Demasiadas peticiones, intenta más tarde'
});

// Rate limiting más estricto para rutas admin
const adminRateLimit = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 20,
  message: 'Límite de peticiones administrativas excedido'
});

// Aplicar rate limiting
router.use(publicRateLimit);
router.use('/admin/*', adminRateLimit);
```

#### 3. **Falta de Sanitización de Headers** - Todas las líneas
**Riesgo:** Medio - Posible exposición de información del servidor

**Solución:**
```js
const helmet = require('helmet');

// Aplicar helmet para seguridad de headers
router.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'"],
      scriptSrc: ["'self'"],
      imgSrc: ["'self'", "data:", "https:"]
    }
  }
}));
```

#### 4. **Orden de Rutas Problemático** - Líneas 13-17
**Riesgo:** Medio - Conflicto potencial entre rutas

```js
// PROBLEMÁTICO - /:slug puede capturar 'seo', 'admin', etc.
router.get('/seo/:slug', CategoryController.getCategorySEO);
router.get('/:slug', CategoryController.getCategoryBySlug); // Esta puede interceptar la anterior
```

**Solución:**
```js
// Rutas específicas ANTES de las genéricas
router.get('/', CategoryController.getAllCategories);
router.get('/menu', CategoryController.getMenuCategories);
router.get('/sitemap', CategoryController.getSitemapData);
router.get('/stats', CategoryController.getCategoryStats);
router.get('/seo/:slug', validateSlug, CategoryController.getCategorySEO);

// Rutas admin (protegidas)
router.get('/admin/all', adminAuth, adminRateLimit, CategoryController.getAllCategoriesAdmin);
router.put('/admin/:slug/toggle', adminAuth, adminRateLimit, validateSlug, CategoryController.toggleCategory);

// Rutas genéricas AL FINAL
router.get('/:slug', validateSlug, CategoryController.getCategoryBySlug);
router.get('/:slug/subcategories', validateSlug, CategoryController.getSubcategories);
```

### **Código Completo Corregido:**

```js
// routes/categories.js - Versión segura
const express = require('express');
const rateLimit = require('express-rate-limit');
const helmet = require('helmet');
const { param, validationResult } = require('express-validator');
const CategoryController = require('../controllers/categoryController');
const { adminAuth } = require('../middleware/auth');

const router = express.Router();

// Seguridad de headers
router.use(helmet({
  contentSecurityPolicy: {
    directives: {
      defaultSrc: ["'self'"],
      styleSrc: ["'self'", "'unsafe-inline'"],
      scriptSrc: ["'self'"],
      imgSrc: ["'self'", "data:", "https:"]
    }
  }
}));

// Rate limiting
const publicRateLimit = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 100,
  message: { error: 'Demasiadas peticiones, intenta más tarde' }
});

const adminRateLimit = rateLimit({
  windowMs: 15 * 60 * 1000,
  max: 20,
  message: { error: 'Límite de peticiones administrativas excedido' }
});

// Validación de slug
const validateSlug = [
  param('slug')
    .isSlug()
    .isLength({ min: 1, max: 100 })
    .matches(/^[a-z0-9-]+$/)
    .withMessage('Slug inválido'),
  (req, res, next) => {
    const errors = validationResult(req);
    if (!errors.isEmpty()) {
      return res.status(400).json({ errors: errors.array() });
    }
    next();
  }
];

// Aplicar rate limiting público
router.use(publicRateLimit);

// Rutas públicas específicas (ORDEN IMPORTANTE)
router.get('/', CategoryController.getAllCategories);
router.get('/menu', CategoryController.getMenuCategories);
router.get('/sitemap', CategoryController.getSitemapData);
router.get('/stats', CategoryController.getCategoryStats);
router.get('/seo/:slug', validateSlug, CategoryController.getCategorySEO);

// Rutas administrativas
router.use('/admin/*', adminRateLimit);
router.get('/admin/all', adminAuth, CategoryController.getAllCategoriesAdmin);
router.put('/admin/:slug/toggle', adminAuth, validateSlug, CategoryController.toggleCategory);

// Rutas públicas genéricas (AL FINAL)
router.get('/:slug', validateSlug, CategoryController.getCategoryBySlug);
router.get('/:slug/subcategories', validateSlug, CategoryController.getSubcategories);

module.exports = router;
```
